     1→// ============================================================================
     2→// Foreign-Key Registry — declarative single-source-of-truth for every
     3→// cross-collection reference in the Urban Castle data model.
     4→// ============================================================================
     5→// Each `ForeignKeyRule` describes one FK:
     6→//   - which collection owns the field (the "child")
     7→//   - which field holds the reference
     8→//   - which collection the field points to (the "parent")
     9→//   - the onDelete policy: cascade | restrict | nullify | ignore
    10→//   - whether the field is allowed to be empty (nullable)
    11→//
    12→// The registry is consumed by:
    13→//   - checker.ts (orphan detection)
    14→//   - cascade.ts (cascade-delete planning)
    15→//   - repair.ts (auto-fix of orphans)
    16→//   - the Integrity UI (FK registry browser)
    17→//
    18→// POLICIES (chosen to mirror real-world construction accounting):
    19→//   - cascade   : child is meaningless without parent → delete with parent
    20→//   - restrict  : parent cannot be deleted while children exist (block)
    21→//   - nullify   : child can survive losing the parent → set field to undefined
    22→//   - ignore    : polymorphic reference (parent collection varies) — handled
    23→//                 elsewhere (e.g. validateBusinessData's assertThreadParentExists)
    24→//
    25→// Polymorphic fields:
    26→//   - tasks/followups/actions: `linked_record_id` + `linked_record_type` — the
    27→//     parent collection depends on the type. Marked "ignore" here because the
    28→//     generic checker cannot resolve the target collection. validateBusinessData
    29→//     covers these via `assertCustomerRelation`.
    30→//   - threads: `kind` + `record_id` — same pattern, covered by
    31→//     `assertThreadParentExists`.
    32→//   - recurringTasks: polymorphic `linked_entity` — ignored.
    33→//   - entityFileAttachments / entityReferenceAssignments: polymorphic
    34→//     `entity_type` + `entity_id` — ignored.
    35→//
    36→// IMPORTANT: this registry COMPLEMENTS validateBusinessData() — it does NOT
    37→// replace it. The business-rules engine still runs on every commit. This
    38→// registry drives the integrity dashboard, cascade-delete planner, and
    39→// on-demand repair.
    40→// ============================================================================
    41→

   400→/** Return every FK rule whose child collection is `collection`. */
   401→export function fksForCollection(collection: string): ForeignKeyRule[] {
   402→    return FOREIGN_KEYS.filter((rule) => rule.collection === collection);
   403→}
   404→
   405→/** Return every FK rule whose parent collection is `collection`
   406→ *  (i.e. the rules that govern what happens when a row in `collection`
   407→ *  is deleted). Used by the cascade-delete planner. */
   408→export function fksTargetingCollection(collection: string): ForeignKeyRule[] {
   409→    return FOREIGN_KEYS.filter((rule) => rule.targetCollection === collection);
   410→}
   411→
   412→/** Return the set of all parent collections referenced by at least one FK. */
   413→export function parentCollections(): string[] {
   414→    const set = new Set<string>();
   415→    for (const rule of FOREIGN_KEYS) {
   416→        if (rule.targetCollection !== "polymorphic") {
   417→            set.add(rule.targetCollection);
   418→        }
   419→    }
   420→    return Array.from(set);
   421→}
   422→
   423→/** Return the set of all child collections that own at least one FK. */
   424→export function childCollections(): string[] {
   425→    const set = new Set<string>();
   426→    for (const rule of FOREIGN_KEYS) {
   427→        set.add(rule.collection);
   428→    }
   429→    return Array.from(set);
   430→}
