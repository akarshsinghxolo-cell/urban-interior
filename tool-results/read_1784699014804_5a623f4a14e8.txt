   332→// updateContractorBid store action. Lets the user revise quote_amount,
   333→// estimated_days, with_material, and evaluation_notes on a submitted bid, and
   334→// also provides a "Withdraw bid" action that sets status="withdrawn"
   335→// (previously unreachable from any UI — see analysis F.5 + F.9).
   336→function EditContractorBidDialog({ bid, onClose, onSave, onWithdraw }: {
   337→    bid: any;
   338→    onClose: () => void;
   339→    onSave: (patch: Partial<any>) => void;
   340→    onWithdraw: () => void;
   341→}) {
   342→    const [quoteAmount, setQuoteAmount] = React.useState<string>(bid.quote_amount != null ? String(bid.quote_amount) : "");
   343→    const [estimatedDays, setEstimatedDays] = React.useState<string>(bid.estimated_days != null ? String(bid.estimated_days) : "");
   344→    const [withMaterial, setWithMaterial] = React.useState<boolean>(Boolean(bid.with_material));
   345→    const [evaluationNotes, setEvaluationNotes] = React.useState<string>(bid.evaluation_notes || "");
   346→    const handleSave = () => {
   347→        const patch: Partial<any> = {};
   348→        const q = parseFloat(quoteAmount);
   349→        if (Number.isFinite(q) && q >= 0 && q !== bid.quote_amount) patch.quote_amount = q;
   350→        const d = parseInt(estimatedDays);
   351→        if (Number.isFinite(d) && d >= 0 && d !== bid.estimated_days) patch.estimated_days = d;
   352→        if (withMaterial !== Boolean(bid.with_material)) patch.with_material = withMaterial;
   353→        if (evaluationNotes.trim() !== (bid.evaluation_notes || "")) patch.evaluation_notes = evaluationNotes.trim();
   354→        if (Object.keys(patch).length === 0) {
   355→            toast.info("No changes to save.");
   356→            return;
   357→        }
   358→        onSave(patch);
   359→    };
   360→    return (<Dialog open onOpenChange={(o) => !o && onClose()}>
   361→      <DialogContent className="max-w-md gap-0 p-0">
   362→        <DialogHeader className="border-b border-border px-5 py-3">
   363→          <DialogTitle className="flex items-center gap-2 text-base"><Pencil className="h-4 w-4 text-primary"/> Edit bid · {bid.bid_no}</DialogTitle>
   364→          <DialogDescription className="text-xs">{bid.contractor_name} · {bid.work_order_no} · {bid.scope}</DialogDescription>
   365→        </DialogHeader>
   366→        <div className="space-y-3 px-5 py-4">
   367→          <div>
   368→            <label className="text-[10px] font-semibold uppercase text-muted-foreground">Quote amount (₹)</label>
   369→            <Input type="number" min="0" step="0.01" value={quoteAmount} onChange={(e) => setQuoteAmount(e.target.value)} placeholder="e.g. 58000" className="h-9 text-sm" autoFocus/>
   370→            <p className="mt-1 text-[10px] text-muted-foreground">Required to be greater than 0 before this bid can be awarded (CV-1 / CV-14 guard in selectContractorBid).</p>
   371→          </div>
   372→          <div>
   373→            <label className="text-[10px] font-semibold uppercase text-muted-foreground">Estimated days</label>
   374→            <Input type="number" min="0" step="1" value={estimatedDays} onChange={(e) => setEstimatedDays(e.target.value)} placeholder="e.g. 7" className="h-9 text-sm"/>
   375→          </div>
   376→          <label className="flex items-center gap-2 text-xs font-medium text-muted-foreground">
   377→            <input type="checkbox" checked={withMaterial} onChange={(e) => setWithMaterial(e.target.checked)} className="h-4 w-4 rounded border-border"/>
   378→            Contractor supplies material (with_material)
   379→          </label>
   380→          <div>
   381→            <label className="text-[10px] font-semibold uppercase text-muted-foreground">Evaluation notes</label>
   382→            <Textarea value={evaluationNotes} onChange={(e) => setEvaluationNotes(e.target.value)} placeholder="Internal notes about this bid — negotiation, references, concerns, etc." rows={3} className="text-sm"/>
   383→          </div>
   384→        </div>
   385→        <DialogFooter className="border-t border-border px-5 py-3">
   386→          <Button variant="outline" size="sm" onClick={onClose}><X className="mr-1 h-3.5 w-3.5"/> Cancel</Button>
   387→          <Button variant="ghost" size="sm" onClick={onWithdraw} className="text-destructive hover:bg-destructive/10" title="Mark this bid as withdrawn — the contractor is no longer competing for this scope.">
   388→            Withdraw bid
   389→          </Button>
   390→          <Button size="sm" onClick={handleSave}>
   391→            <CheckCircle2 className="mr-1 h-3.5 w-3.5"/> Save changes
   392→          </Button>
   393→        </DialogFooter>
   394→      </DialogContent>
   395→    </Dialog>);
   396→}
