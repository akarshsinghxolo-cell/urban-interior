  1099→        // J: Recompute a contractor's reliability_score, on_time_pct, and
  1100→        // rating from actual RA-bill + payment performance + execution-log
  1101→        // on-time filing. Previously these were static master fields set at
  1102→        // contractor creation and never updated — so a contractor with 50
  1103→        // settled RA bills kept whatever score was entered at creation. Now
  1104→        // we derive a 0-100 score from:
  1105→        //   • on-time completion: work orders whose actual_end <= expected_end
  1106→        //   • RA-bill settled vs disputed rate
  1107→        //   • disputed-bill penalty
  1108→        // We write the recomputed fields back to the contractor master so the
  1109→        // ContractorPerformanceModule leaderboard reflects reality.
  1110→        recomputeContractorPerformance: (contractorId) => {
  1111→            const state = get();
  1112→            const actor = state.currentUser();
  1113→            const contractor = state.db.master.contractors.find((c: any) => c.id === contractorId);
  1114→            if (!contractor)
  1115→                throw new Error("Contractor not found.");
  1116→            const contractorWOs = state.db.workOrders.filter((wo: any) => wo.contractor_id === contractorId);
  1117→            // On-time completion: completed WOs with both dates set.
  1118→            const completedWOs = contractorWOs.filter((wo: any) => wo.actual_end && wo.expected_end);
  1119→            const onTimeWOs = completedWOs.filter((wo: any) => wo.actual_end <= wo.expected_end).length;
  1120→            const onTimePct = completedWOs.length > 0
  1121→                ? Math.round((onTimeWOs / completedWOs.length) * 100)
  1122→                : 0;
  1123→            // RA-bill settled vs disputed rate.
  1124→            const contractorBills = state.db.contractorBills.filter((b: any) => b.contractor_id === contractorId && b.status !== "held");
  1125→            const settledBills = contractorBills.filter((b: any) => b.status === "paid" || b.status === "partly_paid").length;
  1126→            const disputedBills = contractorBills.filter((b: any) => b.status === "disputed").length;
  1127→            const settleRate = contractorBills.length > 0
  1128→                ? Math.round((settledBills / contractorBills.length) * 100)
  1129→                : 100;
  1130→            // Composite reliability score.
  1131→            const disputePenalty = Math.min(30, disputedBills * 10);
  1132→            const reliabilityScore = Math.max(0, Math.min(100, Math.round(onTimePct * 0.55 + settleRate * 0.45) - disputePenalty));
  1133→            const rating = reliabilityScore >= 90 ? 5
  1134→                : reliabilityScore >= 75 ? 4
  1135→                    : reliabilityScore >= 60 ? 3
  1136→                        : reliabilityScore >= 40 ? 2
  1137→                            : 1;
  1138→            commitState((s: any) => ({
  1139→                db: {
  1140→                    ...s.db,
  1141→                    master: {
  1142→                        ...s.db.master,
  1143→                        contractors: s.db.master.contractors.map((c: any) => c.id === contractorId
  1144→                            ? {
  1145→                                ...c,
  1146→                                reliability_score: reliabilityScore,
  1147→                                on_time_pct: onTimePct,
  1148→                                rating,
  1149→                                performance_recomputed_at: nowIso(),
  1150→                            }
  1151→                            : c),
  1152→                    },
  1153→                },
  1154→            }));

  1156→                actor: actor.name,
  1157→                actor_role: actor.role,
  1158→                action: `Recomputed contractor performance for ${contractor.name}: reliability=${reliabilityScore}, on-time=${onTimePct}%, rating=${rating} (${contractorWOs.length} WOs, ${contractorBills.length} bills, ${disputedBills} disputed)`,
