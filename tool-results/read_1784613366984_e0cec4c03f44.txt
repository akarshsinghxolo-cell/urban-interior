    77→function lineItemScope(db: any, context: EntityContext) {
    78→    const fromItems = (items: any[] = []) => ({
    79→        articleIds: unique(items.map((item) => item.article_id)),
    80→        variantIds: unique(items.map((item) => item.variant_id)),
    81→        categoryIds: unique(items.map((item) => item.category_id)),
    82→        subcategoryIds: unique(items.map((item) => item.work_required_id)),
    83→    });
    84→    if (context.entityType === "workRequired") {
    85→        const work = db.workRequired.find((row: any) => row.id === context.entityId);
    86→        return { articleIds: [], variantIds: [], categoryIds: unique([work?.work_category_id]), subcategoryIds: unique([work?.work_subcategory_id]) };
    87→    }
    88→    if (context.entityType === "quotation" || context.entityType === "quotation_item") {
    89→        const quotation = db.quotations.find((row: any) => row.id === context.quotationId);
    90→        return fromItems(quotation?.scope_lines || quotation?.items || []);
    91→    }
    92→    if (context.entityType === "workOrder") {
    93→        const workOrder = db.workOrders.find((row: any) => row.id === context.workOrderId);
    94→        const quotation = db.quotations.find((row: any) => workOrder?.quotation_ids?.includes(row.id));
    95→        return fromItems(quotation?.scope_lines || quotation?.items || []);
    96→    }
    97→    if (context.entityType === "boq" || context.entityType === "boq_item") {
    98→        const boq = db.boqs.find((row: any) => row.id === context.entityId || row.items?.some((item: any) => item.id === context.entityId));
    99→        return fromItems(boq?.items || []);
   100→    }
   101→    if (context.entityType === "purchase_order") {
   102→        return fromItems(db.purchaseOrders.find((row: any) => row.id === context.purchaseOrderId)?.items || []);
   103→    }
   104→    if (context.entityType === "grn") {
   105→        return fromItems(db.grns.find((row: any) => row.id === context.grnId)?.items || []);
   106→    }
   107→    if (context.entityType === "vendor_bill") {
   108→        const po = db.purchaseOrders.find((row: any) => row.id === context.purchaseOrderId);
   109→        return fromItems(po?.items || []);
   110→    }
   111→    if (context.entityType === "dispatch") {
   112→        return fromItems(db.dispatches.find((row: any) => row.id === context.entityId)?.items || []);
   113→    }
   114→    if (context.entityType === "inventory") {
   115→        const item = db.inventory.find((row: any) => row.id === context.entityId);
   116→        return { ...fromItems([item]), articleIds: unique([item?.article_id]), variantIds: unique([item?.variant_id]), categoryIds: unique([item?.category_id]), subcategoryIds: unique([item?.work_required_id]) };
   117→    }
   118→    if (context.entityType === "execution_log") {
   119→        const log = db.executionLogs.find((row: any) => row.id === context.entityId);
   120→        return { articleIds: unique((log?.materials_used || []).map((item: any) => item.article_id)), variantIds: [], categoryIds: [], subcategoryIds: [] };
   121→    }
   122→    if (context.entityType === "task") {
   123→        const quotation = db.quotations.find((row: any) => row.id === context.quotationId);
   124→        return fromItems(quotation?.scope_lines || quotation?.items || []);
   125→    }
   126→    if (context.entityType === "payment" || context.entityType === "invoice") {
   127→        const quotation = db.quotations.find((row: any) => row.id === context.quotationId);
   128→        return fromItems(quotation?.scope_lines || quotation?.items || []);
   129→    }
   130→    if (context.entityType === "vendor_rate") {
   131→        const rate = db.master.vendorRates.find((row: any) => row.id === context.entityId);
   132→        const scope = db.master.subcategoryArticleMap.find((row: any) => row.id === rate?.work_required_article_id);
   133→        const work = db.master.workSubcategories.find((row: any) => row.id === scope?.work_required_id);
   134→        return { articleIds: unique([rate?.article_id, scope?.article_id]), variantIds: unique([rate?.variant_id]), categoryIds: unique([work?.category_id]), subcategoryIds: unique([scope?.work_required_id]) };
   135→    }
   136→    return { articleIds: [], variantIds: [], categoryIds: [], subcategoryIds: [] };
   137→}
   138→function entityContext(db: any, entityType: FileAttachmentEntityType, entityId: string): Context {
   139→    const empty: Context = { articleIds: [], variantIds: [], categoryIds: [], subcategoryIds: [] };
   140→    try {
   141→        const resolved = resolveEntityContext(db, entityType, entityId, "Operational media");
   142→        const customer = resolved.customerId ? db.customers.find((item: any) => item.id === resolved.customerId) : undefined;
   143→        const scope = lineItemScope(db, resolved);
   144→        return {
   145→            ...empty,
   146→            ...scope,
   147→            customerId: resolved.customerId,
   148→            customerName: customer?.name,
   149→            customerPhone: customer?.whatsapp || customer?.phone,
   150→            workRequiredId: resolved.workRequiredId,
   151→            quotationId: resolved.quotationId,
   152→            workOrderId: resolved.workOrderId,
   153→            siteId: resolved.siteId,
   154→            roomId: resolved.areaId,
   155→            vendorId: resolved.vendorId,
   156→        };
   157→    }
   158→    catch {
   159→        return empty;
   160→    }
   161→}
   162→function matchesScope(resource: any, context: Context) {
   163→    let score = 0;
   164→    const hasScope = Boolean(resource.category_id || resource.subcategory_id || resource.article_id || resource.variant_id);
   165→    if (resource.variant_id) {
   166→        if (!context.variantIds.includes(resource.variant_id))
   167→            return -1;
   168→        score += 40;
   169→    }
   170→    if (resource.article_id) {
   171→        if (!context.articleIds.includes(resource.article_id))
   172→            return -1;
   173→        score += 30;
   174→    }
   175→    if (resource.subcategory_id) {
   176→        if (!context.subcategoryIds.includes(resource.subcategory_id))
   177→            return -1;
   178→        score += 15;
   179→    }
   180→    if (resource.category_id) {
   181→        if (!context.categoryIds.includes(resource.category_id))
   182→            return -1;
   183→        score += 10;
   184→    }
   185→    return hasScope ? score : 1;
   186→}
   187→function resolveResource(db: any, assignment: EntityReferenceAssignment): ResourceCandidate | null {
   188→    if (assignment.resource_type === "catalogue") {
   189→        const item = db.master.catalogues.find((row: any) => row.id === assignment.resource_id);
   190→        const file = item?.drive_asset_id ? db.master.fileAssets.find((row: any) => row.id === item.drive_asset_id) : undefined;
   191→        if (!item || item.status !== "active")
   192→            return null;
   193→        return { type: "catalogue", id: item.id, title: item.title, url: file?.web_view_link || item.catalog_url, sendable: item.sendable_to_customer !== false, score: 0, subtitle: item.catalog_type || "Catalogue", fileId: file?.id };
   194→    }
   195→    if (assignment.resource_type === "pinterest_board") {
   196→        const item = db.master.pinterestBoards.find((row: any) => row.id === assignment.resource_id);
   197→        if (!item || item.status !== "active")
   198→            return null;
   199→        return { type: "pinterest_board", id: item.id, title: item.title, url: item.board_url, sendable: item.sendable_to_customer !== false, score: 0, subtitle: "Pinterest board" };
   200→    }
   201→    const item = db.master.referenceMedia.find((row: any) => row.id === assignment.resource_id);
   202→    const file = item?.drive_asset_id ? db.master.fileAssets.find((row: any) => row.id === item.drive_asset_id) : undefined;
   203→    if (!item || item.status !== "active")
   204→        return null;
   205→    return { type: "reference_media", id: item.id, title: item.title, url: file?.web_view_link || item.media_url, sendable: item.sendable_to_customer !== false, score: 0, subtitle: "Reference media", fileId: file?.id };
   206→}

   224→    const [fileDraft, setFileDraft] = React.useState({ name: "", url: "", role: "document" as FileAttachmentRole, visibility: "internal", customerShareable: false, tags: "" });
   225→    const context = React.useMemo(() => entityContext(db, entityType, entityId), [db, entityType, entityId]);
   226→    const attachments = (db.entityFileAttachments || [])
   227→        .filter((item) => item.entity_type === entityType && item.entity_id === entityId)
   228→        .map((item) => ({ attachment: item, file: db.master.fileAssets.find((file) => file.id === item.file_asset_id) }))
   229→        .filter((item) => item.file?.status === "active");
   230→    const assignments = (db.entityReferenceAssignments || [])
   231→        .filter((item) => item.entity_type === entityType && item.entity_id === entityId && item.status === "active")
   232→        .map((item) => ({ assignment: item, resource: resolveResource(db, item) }))
   233→        .filter((item): item is {
   234→        assignment: EntityReferenceAssignment;
   235→        resource: ResourceCandidate;
   236→    } => Boolean(item.resource));
   237→    const candidates = React.useMemo(() => {
   238→        const already = new Set(assignments.map(({ resource }) => `${resource.type}:${resource.id}`));
   239→        const output: ResourceCandidate[] = [];

   242→            if (catalogue.status !== "active")

   246→            const exact = relevantLinks.reduce((best: number, link: any) => {
   247→                if (!context.articleIds.includes(link.article_id))
   248→                    return best;
   249→                if (link.variant_id && !context.variantIds.includes(link.variant_id))
   250→                    return best;
   251→                if (link.vendor_id && context.vendorId && link.vendor_id !== context.vendorId)
   252→                    return best;
   253→                return Math.max(best, 30 + (link.variant_id ? 20 : 0) + (link.vendor_id && link.vendor_id === context.vendorId ? 10 : 0));
   254→            }, 0);
   255→            if (!exact)
   256→                continue;
   257→            const candidate = { type: "catalogue" as const, id: catalogue.id, title: catalogue.title, url: file?.web_view_link || catalogue.catalog_url, sendable: catalogue.sendable_to_customer !== false, score: exact, subtitle: catalogue.catalog_type || "Catalogue", fileId: file?.id };
   258→            if (!already.has(`${candidate.type}:${candidate.id}`))
   259→                output.push(candidate);
   260→        }
   261→        for (const board of db.master.pinterestBoards || []) {
   262→            if (board.status !== "active")
   263→                continue;
   264→            const score = matchesScope(board, context);
   265→            if (score < 0)
   266→                continue;
   267→            const candidate = { type: "pinterest_board" as const, id: board.id, title: board.title, url: board.board_url, sendable: board.sendable_to_customer !== false, score, subtitle: "Pinterest board" };
   268→            if (!already.has(`${candidate.type}:${candidate.id}`))
   269→                output.push(candidate);
   270→        }
   271→        for (const media of db.master.referenceMedia || []) {
   272→            if (media.status !== "active")
   273→                continue;
   274→            const score = matchesScope(media, context);
   275→            if (score < 0)
   276→                continue;
   277→            const file = media.drive_asset_id ? db.master.fileAssets.find((row) => row.id === media.drive_asset_id) : undefined;
   278→            const candidate = { type: "reference_media" as const, id: media.id, title: media.title, url: file?.web_view_link || media.media_url, sendable: media.sendable_to_customer !== false, score, subtitle: "Reference media", fileId: file?.id };
   279→            if (!already.has(`${candidate.type}:${candidate.id}`))
   280→                output.push(candidate);
   281→        }
   282→        return output.sort((a, b) => b.score - a.score || a.title.localeCompare(b.title)).slice(0, 12);

   308→            customer_id: context.customerId,
   309→            work_required_id: context.workRequiredId,
   310→            quotation_id: context.quotationId,
   311→            work_order_id: context.workOrderId,
   312→            site_id: context.siteId,
   313→            area_id: context.roomId,
   314→            article_id: context.articleIds[0],
