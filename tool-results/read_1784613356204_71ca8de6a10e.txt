   315→/**
   316→ * Detect whether the cursor is inside an active @[mention trigger.
   317→ * Returns the anchor (index of `@`) and the query text after `[` if active,
   318→ * or null if the cursor is not inside a mention trigger.
   319→ */
   320→function detectMentionTrigger(text: string, cursor: number): { anchor: number; query: string } | null {
   321→    if (cursor < 2) return null;
   322→    // Walk backwards from the cursor looking for `@[`.
   323→    let i = cursor - 1;
   324→    while (i >= 1) {
   325→        const ch = text[i];
   326→        if (ch === "\n") return null; // crossed a line — not a mention
   327→        if (ch === "]" ) return null; // mention already closed
   328→        if (ch === "[" && text[i - 1] === "@") {
   329→            const query = text.slice(i + 1, cursor);
   330→            if (query.includes(")") ) return null;
   331→            return { anchor: i - 1, query };
   332→        }
   333→        i--;
   334→    }
   335→    return null;
   336→}

   338→/** Filter thread messages by search text + kind filter. Preserves nesting
   339→ *  (a parent is kept if it matches OR any of its descendants match). */
   340→function filterThreadMessages(messages: ThreadMessage[], search: string, filter: "all" | "comment" | "decision" | "system" | "alert"): ThreadMessage[] {
   341→    const q = search.trim().toLowerCase();
   342→    const kindMatch = (m: ThreadMessage): boolean => {
   343→        if (filter === "all") return true;
   344→        if (filter === "comment") return m.kind === "comment" || m.kind === "proof";
   345→        return m.kind === filter;
   346→    };
   347→    const textMatch = (m: ThreadMessage): boolean => {
   348→        if (!q) return true;
   349→        return (m.body || "").toLowerCase().includes(q) || (m.author_name || "").toLowerCase().includes(q);
   350→    };
   351→    // Build child→parent index so we can keep ancestors of matching descendants.
   352→    const byParent = new Map<string | undefined, ThreadMessage[]>();
   353→    messages.forEach((m) => {
   354→        const key = m.parent_message_id;
   355→        byParent.set(key, [...(byParent.get(key) || []), m]);
   356→    });
   357→    const keep = new Set<string>();
   358→    const walk = (m: ThreadMessage): boolean => {
   359→        const selfMatch = kindMatch(m) && textMatch(m);
   360→        const children = byParent.get(m.id) || [];
   361→        const childMatch = children.some(walk);
   362→        if (selfMatch || childMatch) keep.add(m.id);
   363→        return selfMatch || childMatch;
   364→    };
   365→    const roots = byParent.get(undefined) || [];
   366→    roots.forEach(walk);
   367→    return messages.filter((m) => keep.has(m.id));
   368→}

   375→function ThreadTree({ messages, onReply }: {
   376→    messages: ThreadMessage[];
   377→    onReply: (messageId: string) => void;
   378→}) {
   379→    const byParent = React.useMemo(() => {
   380→        const map = new Map<string | undefined, ThreadMessage[]>();
   381→        messages.forEach((message) => {
   382→            const key = message.parent_message_id;
   383→            map.set(key, [...(map.get(key) || []), message]);
   384→        });
   385→        return map;
   386→    }, [messages]);
   387→    const render = (message: ThreadMessage, depth = 0): React.ReactNode => (<div key={message.id} className={depth ? "ml-4 border-l border-primary/20 pl-3" : ""}>
   388→      <MessageBubble m={message} nested={depth > 0} onReply={() => onReply(message.id)}/>
   389→      <div className="mt-2 space-y-2">{(byParent.get(message.id) || []).map((child) => render(child, depth + 1))}</div>
   390→    </div>);
   391→    const roots = byParent.get(undefined) || [];
   392→    return <>{roots.map((message) => render(message))}</>;
   393→}
   394→/** Resolve a ThreadMessageAttachment to a FilePreviewSource using the db. */
   395→function attachmentToPreviewSource(db: RDashDatabase, att: ThreadMessageAttachment): FilePreviewSource | null {
   396→    if (att.entity_file_attachment_id) {
   397→        const attached = attachedFileById(db, att.entity_file_attachment_id);
   398→        if (attached)
   399→            return assetPreview(attached.asset);
   400→    }
   401→    if (att.file_asset_id) {
   402→        const asset = db.master.fileAssets.find((a: FileAsset) => a.id === att.file_asset_id);
   403→        if (asset)
   404→            return assetPreview(asset);
   405→    }
   406→    if (att.thumbnail_url || att.preview_url) {
   407→        return {
   408→            fileName: att.name,
   409→            mimeType: att.mime,
   410→            url: att.preview_url,
   411→            thumbnailUrl: att.thumbnail_url,
   412→        };
   413→    }
   414→    return null;
   415→}

   466→function MessageBubble({ m, nested = false, onReply }: {
   467→    m: ThreadMessage;
   468→    nested?: boolean;
   469→    onReply: () => void;
   470→}) {
   471→    const db = useRDashStore((state) => state.db);
   472→    const proof = attachedPreview(db, m.proof_attachment_id);
   473→    const isSystem = m.kind === "system";
   474→    const isAlert = m.kind === "alert";
   475→    const isDecision = m.kind === "decision";
   476→    const isProof = m.kind === "proof";
   477→    const absTime = formatDate(m.created_at);
   478→    // System + alert messages render as a centered timeline pill — no avatar.
   479→    // Alerts (mention backlinks) get a primary accent; system events get muted.
   480→    if (isSystem || isAlert)
   481→        return (<div className="flex items-center justify-center py-0.5">
   482→        <span title={absTime} className={cn("inline-flex items-center gap-1.5 rounded-full border px-3 py-1 text-[11px] font-medium", isAlert ? "border-primary/20 bg-primary/[0.07] text-primary" : "border-border bg-muted/50 text-muted-foreground")}>
   483→          {isAlert ? <ArrowUpRight className="h-3 w-3"/> : <Info className="h-3 w-3"/>}{m.body}<span className="text-[10px] opacity-60">· {relativeDay(m.created_at)}</span>
   484→        </span>
   485→      </div>);
   486→    // Comment / decision / proof messages render as a full bubble with avatar.
   487→    return (<div className={cn("group flex gap-2.5 rounded-xl p-1.5 transition-colors hover:bg-muted/30", nested && "ml-4 border-l-2 border-primary/15 pl-3 hover:bg-primary/[0.03]", isDecision && "border border-success/20 bg-success/[0.04]", isProof && "border border-primary/15 bg-primary/[0.02]")}>
   488→      <Avatar name={m.author_name} size={30}/>
   489→      <div className="min-w-0 flex-1">
   490→        <div className="flex items-center gap-1.5">
   491→          <span className="text-xs font-semibold text-foreground">{m.author_name}</span>
   492→          {m.author_role && <span className="rounded bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground">{m.author_role}</span>}
   493→          {isDecision && <span className="inline-flex items-center gap-1 rounded-full bg-success/15 px-1.5 py-0.5 text-[10px] font-semibold text-success"><CheckCircle2 className="h-3 w-3"/>Decision</span>}
   494→          {isProof && <span className="inline-flex items-center gap-1 rounded-full bg-primary/10 px-1.5 py-0.5 text-[10px] font-semibold text-primary"><Paperclip className="h-3 w-3"/>Proof</span>}
   495→          <span title={absTime} className="inline-flex items-center gap-0.5 text-[10px] text-muted-foreground/80"><Clock className="h-2.5 w-2.5 opacity-60"/>{relativeDay(m.created_at)}</span>
   496→        </div>
   497→        <MentionBody body={m.body}/>
   498→        {m.attachments && m.attachments.length > 0 && <MessageAttachments attachments={m.attachments}/>}
   499→        {proof && <FilePreview file={proof} compact controls className="mt-1.5 max-w-xs"/>}
   500→        <button type="button" onClick={onReply} className="mt-1 text-[10px] font-medium text-primary opacity-0 transition-opacity group-hover:opacity-100 hover:underline">Reply</button>
   501→      </div>
   502→    </div>);
   503→}
